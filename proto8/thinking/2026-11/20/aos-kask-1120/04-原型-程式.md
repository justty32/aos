## `kask/answer`

```python
#!/usr/bin/env python3
# Trial before_verdict hook (line aos-kask-1120): when a letter only asks "what do you take?" (body.kernel_ask is
# true; the key name is a trial placeholder), reply with the kernel's declaration file and stop the letter here.
# Mode A (default): send the answer straight to reply_to, then exit 10, so the kernel refuses the letter (that sends
# a second, HookRejected reply). Mode B (env KASK_MODE=endpoint, llm kernel only): rewrite the body into a brain
# request for the trial endpoint "kask-answer", which answers with the declaration as the message content.
import json
import os
import subprocess
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
DECL = os.path.join(HERE, "..", "kask.json")          # the "other json" next to kernel.json (trial place and name)

event = json.load(sys.stdin)
body = event.get("body")
if not (isinstance(body, dict) and body.get("kernel_ask") is True):
    sys.exit(0)                                       # not a question: say nothing, the letter goes on as before

if os.environ.get("KASK_MODE") == "endpoint" or (isinstance(body, dict) and body.get("kask_mode") == "endpoint"):
    print(json.dumps({"body": {"endpoint": "kask-answer", "request": {"messages": []}}}))
    sys.exit(0)

try:
    with open(DECL, encoding="utf-8") as stream:
        answer = {"kernel_ask": json.load(stream)}
except FileNotFoundError:
    answer = {"kernel_ask": None, "why": "this kernel gives no declaration"}
except (OSError, ValueError) as error:
    answer = {"kernel_ask": None, "why": "declaration unreadable: %s" % error}

reply_to = event.get("reply_to")
if isinstance(reply_to, str) and reply_to:
    if not reply_to.startswith("/"):
        reply_to = os.path.abspath(os.path.join(event.get("stack") or ".", "requests", reply_to))
    sent = subprocess.run(["aos-post", "send", "--in-reply-to", event["letter"], reply_to],
                          input=json.dumps(answer), text=True, capture_output=True)
    if sent.returncode != 0:
        print("kask: cannot send the answer: %s" % sent.stderr.strip())
        sys.exit(10)
print("answered by the kask hook (this refusal only closes the question letter)")
sys.exit(10)
```

#!/usr/bin/env python3
# Trial endpoint function (line aos-kask-1120, mode B): answers any brain request with the kernel's declaration file
# (../kask.json, in the kernel definition folder) as the message content; usage is zero, it calls no model.
import json
import os
import sys

json.load(sys.stdin)
here = os.path.dirname(os.path.abspath(__file__))
try:
    with open(os.path.join(here, "..", "kask.json"), encoding="utf-8") as stream:
        text = stream.read()
except OSError:
    text = "null"
print(json.dumps({"message": {"role": "assistant", "content": text},
                  "usage": {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0}, "model": "kask-answer"}))

## `asker.py`

```python
#!/usr/bin/env python3
# Trial asker (line aos-kask-1120): send one letter to a kernel and print every reply that comes back for it.
# usage: asker.py llm|procs TARGET BOX BODY_JSON [WAIT_S]   (llm: TARGET = llm kernel stack, sent with aos-llm send;
#        procs: TARGET = the station's requests mailbox, sent with aos-post send). After the first reply it keeps
#        watching 1.5 s more, to see second replies. Prints one JSON line per reply: ms since send, reply body.
import json
import os
import subprocess
import sys
import time

kind, target, box, body = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4]
wait_s = float(sys.argv[5]) if len(sys.argv) > 5 else 5.0
subprocess.run(["aos-post", "init", box], check=True, capture_output=True)
if kind == "llm-tool":       # through aos-llm send (it checks the body first)
    env = dict(os.environ, AOS_LLM_KERNEL=target)
    sent = subprocess.run(["aos-llm", "send", "--reply-to", box], input=body, text=True, capture_output=True, env=env)
elif kind == "llm":          # plain aos-post into the llm kernel's requests/ (aos-llm refuses a body without request)
    sent = subprocess.run(["aos-post", "send", "--reply-to", box, os.path.join(target, "requests")],
                          input=body, text=True, capture_output=True)
else:
    sent = subprocess.run(["aos-post", "send", "--reply-to", box, target], input=body, text=True, capture_output=True)
t0 = time.monotonic()
if sent.returncode != 0:
    print(json.dumps({"send_failed": sent.stderr.strip()}))
    sys.exit(1)
letter = sent.stdout.strip()
got, first, seen = [], None, set()
end = t0 + wait_s
while time.monotonic() < end:
    for name in sorted(os.listdir(os.path.join(box, "new"))):
        if name in seen:
            continue
        try:
            with open(os.path.join(box, "new", name, "letter.json") if os.path.isdir(os.path.join(box, "new", name))
                      else os.path.join(box, "new", name)) as stream:
                env_ = json.load(stream)
        except (OSError, ValueError):
            continue
        if env_.get("in_reply_to") != letter:
            continue
        seen.add(name)
        ms = round((time.monotonic() - t0) * 1000)
        got.append(ms)
        print(json.dumps({"ms": ms, "reply_sender": env_.get("sender"), "body": env_.get("body")}, ensure_ascii=False))
        if first is None:
            first = time.monotonic()
            end = min(end, first + 1.5)
    time.sleep(0.02)
if not got:
    print(json.dumps({"no_reply_within_s": wait_s, "letter": letter}))
```

## `run2.sh`

```sh
#!/bin/bash
# Second run (line aos-kask-1120): the llm kernel cases again, letters sent with plain aos-post because aos-llm send
# refuses a body without "request" (run1 found that); also a burst on the procs station and a stopped kernel.
. /home/guanyu/.aos/scratch/aos-kask-1120/env.sh
T=$S/t
L=$T/llm/stacks/llm
K=$T/pk/st/requests
B=$S/boxes2
mkdir -p $B
cd $S
ask() { echo "--- $1: $2 -> $3"; python3 $S/asker.py $2 $( [ ${2%-tool} = llm ] && echo $L || echo $K ) $B/$1 "$3" ${4:-5}; }
use() { cp $T/$1/kernel.$2.json $T/$1/def/kernel.json; sleep 0.8; echo "=== $1 kernel.json = $2"; }

cp $T/llm/kernel.nohook.json $T/llm/def/kernel.json
(cd $T/llm && setsid aos-exec --stack $L def >> $S/llm.log 2>&1 < /dev/null & echo $! > $S/llm2.pid)
sleep 2; echo "llm restarted, pid $(cat $S/llm2.pid)"

echo "##### 1. no hook (today), plain aos-post"
ask a1 llm '{"kernel_ask": true}'
ask a2 llm '{"procs_state_path": true}'
ask a0 llm-tool '{"kernel_ask": true}'

echo "##### 2. kask hook, mode A"
use llm hook
ask b1 llm '{"kernel_ask": true}'
ask b1x llm '{"kernel_ask": true, "endpoint": "fake", "request": {"messages": [{"role": "user", "content": "both in one"}]}}' 8
echo "b1 leftovers in new/: $(ls $B/b1/new | wc -l)"

echo "##### 3. no declaration file"
mv $T/llm/def/kask.json $T/llm/def/kask.json.off
ask c1 llm '{"kernel_ask": true}'
mv $T/llm/def/kask.json.off $T/llm/def/kask.json

echo "##### 4. mode B: hook turns the question into a brain request for endpoint kask-answer"
ask d1 llm '{"kernel_ask": true, "kask_mode": "endpoint"}' 15
echo "last ledger row:"; tail -1 $L/ledger.jsonl

echo "##### 5. burst: 30 questions at once to each kernel (mode A) plus one normal llm request"
python3 - $L $K $B <<'EOF'
import json, os, subprocess, sys, time
L, K, B = sys.argv[1], sys.argv[2], sys.argv[3]
def burst(tag, box_to, normal_body=None):
    box = os.path.join(B, "burst-" + tag); subprocess.run(["aos-post", "init", box], check=True)
    t0 = time.monotonic(); names = []
    for i in range(30):
        names.append(subprocess.run(["aos-post", "send", "--reply-to", box, box_to], input='{"kernel_ask": true}',
                                    text=True, capture_output=True).stdout.strip())
    normal = None
    if normal_body:
        normal = subprocess.run(["aos-post", "send", "--reply-to", box, box_to], input=normal_body,
                                text=True, capture_output=True).stdout.strip()
    sent_ms = round((time.monotonic() - t0) * 1000)
    want = set(names) | ({normal} if normal else set()); done = {}; count = 0; end = time.monotonic() + 30
    while time.monotonic() < end and len(done) < len(want):
        for n in os.listdir(os.path.join(box, "new")):
            p = os.path.join(box, "new", n, "letter.json")
            try: e = json.load(open(p))
            except Exception: continue
            r = e.get("in_reply_to")
            if r in want and r not in done:
                done[r] = round((time.monotonic() - t0) * 1000)
        time.sleep(0.02)
    time.sleep(1.5)
    q = sorted(v for k, v in done.items() if k != normal)
    print(json.dumps({"kernel": tag, "sent_ms": sent_ms, "answered": len(q), "first_ms": q[0] if q else None,
                      "last_ms": q[-1] if q else None, "normal_request_ms": done.get(normal) if normal else None,
                      "letters_in_box": len(os.listdir(os.path.join(box, "new")))}))
burst("llm", os.path.join(L, "requests"),
      '{"endpoint": "fake", "request": {"messages": [{"role": "user", "content": "x"}]}}')
burst("procs", K)
EOF

echo "##### 6. kernel stopped: ask and wait 3 s"
aos-daemon-ctl -d $L shutdown >/dev/null 2>&1; sleep 1.5
ask e1 llm '{"kernel_ask": true}' 3
echo "letters left in the llm kernel's requests/new: $(ls $L/requests/new | wc -l)"
echo "hook failures / refusals in logs: llm $(grep -c HookFailed $S/llm.log) pk $(grep -c HookFailed $S/pk.log)"
```
