## `coarsen.py`

```python
#!/usr/bin/env python3
# Toy "coarsening" demo for a resource claim: read claims, apply a rules file, print a 0..1 cost or "unreadable".
import json
import re
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent


def dig(obj, path):
    """Return the value at a dotted path, or None when any step is missing."""
    for key in path.split("."):
        if not isinstance(obj, dict) or key not in obj:
            return None
        obj = obj[key]
    return obj


def number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def to_ms(value, unit):
    if number(value):
        return value * {"ms": 1, "s": 1000, "m": 60000}.get(unit or "ms", 0) or None
    if isinstance(value, str):
        match = re.fullmatch(r"\s*([0-9.]+)\s*(ms|s|m)\s*", value)
        if match:
            return float(match.group(1)) * {"ms": 1, "s": 1000, "m": 60000}[match.group(2)]
    return None


def read_tokens(claim, readers):
    """Try each reader in order; return (tokens, why_not)."""
    whys = []
    for reader in readers:
        value = dig(claim, reader["path"])
        if value is None:
            continue
        if reader["kind"] == "number" and number(value):
            return value, None
        if reader["kind"] == "sum_keys" and isinstance(value, dict):
            parts = [value.get(key) for key in reader["keys"] if key in value]
            bad = [part for part in parts if not number(part)]
            if bad:
                whys.append("%s has a non-number part %s (maybe a file path; not opened)" % (reader["path"], json.dumps(bad[0])))
                continue
            if parts:
                return sum(parts), None
        whys.append("%s is %s, no reader fits" % (reader["path"], json.dumps(value)))
    return None, "; ".join(whys) or "no token field found"


def read_ms(claim, readers):
    for reader in readers:
        value = dig(claim, reader["path"])
        if value is None:
            continue
        unit = dig(claim, reader["unit_path"]) if "unit_path" in reader else reader.get("unit")
        ms = to_ms(value, unit)
        if ms is not None:
            return ms, None
    return None, "no time field found"


def coarsen(claim, rules):
    """Return {"cost": float, "how": ...} or {"unreadable": reason}."""
    if not isinstance(claim, dict):
        return {"unreadable": "claim is not an object"}
    direct = claim.get("cost")
    if number(direct) and 0 <= direct <= 1:
        return {"cost": float(direct), "how": "follows the schema"}
    tokens, why_t = read_tokens(claim, rules["token_readers"])
    ms, why_m = read_ms(claim, rules["time_readers"])
    if tokens is None and ms is None:
        return {"unreadable": "%s; %s" % (why_t, why_m)}
    if (tokens is None or ms is None) and rules["missing_part"] == "unreadable":
        return {"unreadable": why_t if tokens is None else why_m}
    full, weight = rules["full"], rules["weights"]
    cost = weight["token"] * (tokens or 0) / full["token"] + weight["ms"] * (ms or 0) / full["ms"]
    return {"cost": round(min(1.0, cost), 4), "how": "rule v%d: tokens=%s ms=%s" % (rules["version"], tokens, ms)}


def main():
    rules = json.loads((HERE / (sys.argv[1] if len(sys.argv) > 1 else "rules.json")).read_text())
    unread = HERE / "unreadable.jsonl"
    for line in (HERE / "claims.jsonl").read_text().splitlines():
        sample = json.loads(line)
        result = coarsen(sample["claim"], rules)
        print("%-3s %-70s -> %s" % (sample["id"], json.dumps(sample["claim"]), json.dumps(result)))
        if "unreadable" in result:
            with unread.open("a") as stream:
                stream.write(json.dumps({"id": sample["id"], "rules_version": rules["version"],
                                         "claim": sample["claim"], "why": result["unreadable"]}) + "\n")


if __name__ == "__main__":
    main()
```

## `refine.py`

```python
#!/usr/bin/env python3
# Toy "refining" demo: a kernel that eats cost:{token,time} gets a coarser claim and splits it, marking guessed values.
import json
from pathlib import Path

HERE = Path(__file__).resolve().parent


def number(value):
    return isinstance(value, (int, float)) and not isinstance(value, bool)


def refine(claim, rules):
    """Return {"cost": {"token", "time"}, "guessed": [...]} or {"unreadable": reason}."""
    cost = claim.get("cost") if isinstance(claim, dict) else None
    if number(cost):
        token = round(cost * rules["split"]["token"] * rules["per_point"]["token"])
        time = round(cost * rules["split"]["time"] * rules["per_point"]["time"])
        return {"cost": {"token": token, "time": time}, "guessed": ["token", "time"],
                "how": "rule v%d: bare number read as points and split" % rules["version"]}
    if isinstance(cost, dict) and (number(cost.get("token")) or number(cost.get("time"))):
        out, guessed = {}, []
        for key in ("token", "time"):
            if number(cost.get(key)):
                out[key] = cost[key]
            else:
                out[key] = rules["default"][key]
                guessed.append(key)
        how = "missing parts filled with defaults" if guessed else "already in my shape"
        return {"cost": out, "guessed": guessed, "how": "rule v%d: %s" % (rules["version"], how)}
    return {"unreadable": "no cost number or cost.token/cost.time found"}


def main():
    rules = json.loads((HERE / "refine_rules.json").read_text())
    for claim in ({"cost": 43}, {"cost": {"token": 13, "time": 123}}, {"cost": {"token": 13}}, {"cost": "a lot"}):
        print("%-34s -> %s" % (json.dumps(claim), json.dumps(refine(claim, rules))))


if __name__ == "__main__":
    main()
```
