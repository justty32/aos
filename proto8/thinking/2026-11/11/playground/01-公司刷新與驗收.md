## 照抄（刷新）

```sh
PG=${PG:-~/code/playground}
( set -eu
R=$HOME/repo/aos-tools/proto8                             # this machine's repo
BAK=$PG.bak-$(date +%m%d-%H%M)
[ -f "$PG/env.sh" ] && [ -f "$PG/root/daemon.json" ] || { echo "not a playground: $PG"; exit 1; }
n=1; while [ -e "$BAK" ]; do n=$((n+1)); BAK=$PG.bak-$(date +%m%d-%H%M)-$n; done   # never overwrite a backup
cd "$HOME"
# 0) stop the tree if its root is running (only the pid in $PG/.stack/state.json whose cwd is $PG/root)
mine() { for p in /proc/[0-9]*; do n=${p#/proc/}; [ "$n" != "$$" ] || continue
  case "$(command cat "$p/comm" 2>/dev/null)" in bash|zsh|sh|dash|fish) continue;; esac
  c=$(readlink "$p/cwd" 2>/dev/null || true); a=$(command cat "$p/cmdline" 2>/dev/null | tr '\0' ' ')
  case "$c/ $a" in "$PG"/*|*" $PG"/*|*"=$PG"/*) echo "$n $a";; esac; done; }
rp=$(sed -n 's/.*"pid": *\([0-9][0-9]*\).*/\1/p' "$PG/.stack/state.json" 2>/dev/null | head -n 1)
if [ -n "$rp" ] && [ "$(readlink "/proc/$rp/cwd" 2>/dev/null || true)" = "$PG/root" ]; then
  i=0; until aos-daemon-ctl -d "$PG/.stack" shutdown; do i=$((i+1)); [ $i -lt 30 ] || exit 1; sleep 2; done
fi
i=0; while [ -n "$(mine)" ]; do i=$((i+1)); [ $i -lt 30 ] || { echo "still running under $PG:"; mine; exit 1; }; sleep 1; done
# 1) back up the whole playground
cp -a "$PG" "$BAK" && echo "backup: $BAK"
cd "$PG"
# 2) the four copies (lib py templates knowledge) := the repo's; templates find them through $AOS_HOME = this playground
rm -rf lib py templates knowledge && cp -r "$R/lib" "$R/py" "$R/templates" "$R/knowledge" .
find lib py templates -name __pycache__ -prune -exec rm -rf {} +
# 3) env.sh: AOS_HOME = the playground itself; the old AOS_MEMORY line -> AOS_KNOWLEDGE (renamed 11-10)
sed -i 's|^export AOS_MEMORY=.*|export AOS_KNOWLEDGE=$PG/knowledge          # knowledge base (knowledge_read / knowledge_write; the definitions also set it per tool)|; s|AOS_MEMORY＝知識庫（memory 工具讀它）|AOS_KNOWLEDGE＝知識庫（knowledge_read／knowledge_write 讀它）|' env.sh
grep -q '^export AOS_HOME=' env.sh || echo 'export AOS_HOME=$PG   # definitions use {"$env": "AOS_HOME", ...}: this playground (lib/ py/ templates/ knowledge/)' >> env.sh
# 4) llm kernel definition := the repo's; put back this machine's stupid-fast model (LM Studio) from the backup; add one scripted fake brain
rm -rf llm_kernel/def && cp -r "$R/kernel/llm-kernel" llm_kernel/def
for k in model summary; do aos-json set "llm_kernel/def/kernel.json#/endpoints/stupid-fast/$k" "$(aos-json get "$BAK/llm_kernel/def/kernel.json#/endpoints/stupid-fast/$k")"; done
aos-json set 'llm_kernel/def/kernel.json#/endpoints/fake-demo-today' '{"exec": {"$env": "AOS_HOME", "$path": "/lib/base/llm_fake"}, "speed": "fast", "cost": "free",
 "summary": "playground fake brain for 11-11: changes its own system text, starts a subagent and asks it, then puts both back",
 "script": [{"tool_calls": [{"name": "self_system", "arguments": {"op": "set", "text": "Answer in one short line."}}]},
            {"tool_calls": [{"name": "self_system", "arguments": {"op": "get", "long": true}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "add", "name": "kid", "def_json": {"$env": "AOS_HOME", "$path": "/templates/agent-fsm"}}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "ask", "name": "kid", "text": "what time is it?"}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "rm", "name": "kid", "purge": true}}]},
            {"tool_calls": [{"name": "self_system", "arguments": {"op": "reset"}}]},
            {"content": "I set my own system text, read it back, started kid, asked it the time, removed it, and reset my system text."}]}'
# 5) agents := fresh templates (eleven packs, 35 tools); eca_bob = claude haiku with its old budget; eca_cody = codex gpt-6-luna, no screen
budget=$(aos-json get "$BAK/agents/eca_bob/cli.json#/print_flags/1" 2>/dev/null || echo '"0.3"')
rm -rf agents/fsm_ann agents/eca_bob agents/eca_cody
cp -r "$R/templates/agent-fsm" agents/fsm_ann
cp -r "$R/templates/eca-claude" agents/eca_bob
cp -r "$R/templates/eca-claude" agents/eca_cody
find agents -mindepth 2 -maxdepth 2 \( -name tests -o -name __pycache__ \) -exec rm -rf {} +
echo '{"model": "haiku"}' > agents/eca_bob/llm.json
aos-json set agents/eca_bob/cli.json#/print_flags/1 "$budget"
echo '{"model": "gpt-6-luna"}' > agents/eca_cody/llm.json
command cat > agents/eca_cody/cli.json <<'EOF'
{"kind": "codex",
 "flags": ["-c", "sandbox_mode=\"read-only\"", "-c", "approval_policy=\"never\""],
 "print_flags": [],
 "tools_flags": ["--knowledge", "knowledge_read"],
 "timeout_ms": 600000}
EOF
rm -rf agents/eca_cody/screen
aos-json del 'agents/eca_cody/interfaces.json#/aos-agent/screen'
aos-json set agents/eca_cody/_metainfo.json#/summary '"Run Codex CLI (gpt-6-luna) as an aos agent: one turn per queue batch, answers appended to history; no screen."'
# 6) clear the images: the three stacks' runtime files (agents' history, mailboxes, brain settings set with llm set)
rm -rf .stack ws/.stack && find llm_kernel -mindepth 1 -maxdepth 1 ! -name def -exec rm -rf {} +
echo "refreshed: $PG (agents: $(command ls agents | tr '\n' ' '))"
)
```

會清掉：三隻的對話、信箱、`llm set`／`tools add`／`hook add`／`system set` 改過的設定、子 agent、總機帳本、試驗場 `knowledge/` 裡 agent 寫的東西。保留：`env.sh` 其他行、`ws/` 裡的檔、`root/`、`daemon/`、eca_bob 的 haiku 與額度、`stupid-fast` 的模型。刷新後三隻沒掛在領班上，要 `aos-agent ctl add ../agents/<名>`（這次驗完已掛好）。
