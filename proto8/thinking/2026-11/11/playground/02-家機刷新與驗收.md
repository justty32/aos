```sh
PG=${PG:-~/code/playground}
( set -eu
R=$HOME/repo/simple_tools/aos/proto8                      # this machine's repo
[ -f "$PG/env.sh" ] && [ -f "$PG/root/daemon.json" ] || { echo "not a playground: $PG"; exit 1; }

mine() { for p in /proc/[0-9]*; do n=${p#/proc/}; [ "$n" != "$$" ] || continue
  case "$(command cat "$p/comm" 2>/dev/null)" in bash|zsh|sh|dash|fish) continue;; esac
  c=$(readlink "$p/cwd" 2>/dev/null || true); a=$(command cat "$p/cmdline" 2>/dev/null | tr '\0' ' ')
  case "$c/ $a" in "$PG"/*|*" $PG"/*|*"=$PG"/*) echo "$n $a";; esac; done; }

if [ -n "$rp" ] && [ "$(readlink "/proc/$rp/cwd" 2>/dev/null || true)" = "$PG/root" ]; then
  i=0; until aos-daemon-ctl -d "$PG/.stack" shutdown; do i=$((i+1)); [ $i -lt 30 ] || exit 1; sleep 2; done
fi
i=0; while [ -n "$(mine)" ]; do i=$((i+1)); [ $i -lt 30 ] || { echo "still running under $PG:"; mine; exit 1; }; sleep 1; done
cd "$PG"
# 1) no backup (user 11-11: the old playground can go). Keep env.sh, root/, daemon/, ws/proj/; everything else is rebuilt.
rm -rf lib py templates knowledge kb agents llm_kernel .stack ws/.stack ws/hooks ws/defs root.log
# 2) the four copies (lib py templates knowledge) := the repo's; templates find them through $AOS_HOME = this playground

# 3) llm kernel definition := the repo's; stupid-fast = this machine's LM Studio gemma; one scripted fake brain for the new tools

aos-json set 'llm_kernel/def/kernel.json#/endpoints/stupid-fast/model' '"google/gemma-4-e4b"'
aos-json set 'llm_kernel/def/kernel.json#/endpoints/stupid-fast/summary' '"LM Studio gemma-4-e4b on this machine (home Manjaro): quick and rough, one at a time"'
aos-json set 'llm_kernel/def/kernel.json#/endpoints/fake-demo-today' '{"exec": {"$env": "AOS_HOME", "$path": "/lib/base/llm_fake"}, "speed": "fast", "cost": "free",
 "summary": "playground fake brain for 11-11: changes its own system text, starts a subagent, asks it and lists it, puts the system back, writes a todo, then does one task with do_task start / peek_now_time / end (folded into a summary)",
 "script": [{"tool_calls": [{"name": "self_system", "arguments": {"op": "set", "text": "Answer in one short line."}}]},
            {"tool_calls": [{"name": "self_system", "arguments": {"op": "get", "long": true}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "add", "name": "kid", "def_json": {"$env": "AOS_HOME", "$path": "/templates/agent-fsm"}}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "ask", "name": "kid", "text": "what time is it?"}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "ls"}}]},
            {"tool_calls": [{"name": "subagent", "arguments": {"op": "rm", "name": "kid", "purge": true}}]},
            {"tool_calls": [{"name": "self_system", "arguments": {"op": "reset"}}]},

            {"tool_calls": [{"name": "do_task", "arguments": {"op": "start", "task": "Check the time"}}]},

mkdir agents
cp -r "$R/templates/agent-fsm" agents/fsm_ann
cp -r "$R/templates/eca-claude" agents/eca_bob
find agents -mindepth 2 -maxdepth 2 \( -name tests -o -name __pycache__ \) -exec rm -rf {} +
echo '{"model": "stupid-fast"}' > agents/fsm_ann/llm.json        # fsm_ann's real brain: LM Studio gemma (llm set model=fake for the fake one)

printf '#!/bin/sh\n# ping_me - hook handler: mail this agent one line "tick: what time is it?" (for on_clock; the agent = last part of the event'"'"'s stack)\ns=$(python3 -c '"'"'import json,sys; print(json.load(sys.stdin)["stack"])'"'"')\nexec aos-agent say --target "${s##*/}" "tick: what time is it?" >/dev/null\n' > ws/hooks/ping_me

echo "refreshed: $PG (agents: $(command ls agents | tr '\n' ' '))"
```

## 驗收（11-11 20:45～20:50，家機）

| 步 | 結果 |
|---|---|
| 刷新前 | 樹停著（`state.json` stopped、pid 不在），沒有行程在試驗場底下 |
| 刷新、起根、三台 ping、掛兩隻 | `refreshed: … (agents: eca_bob fsm_ann)`；根表 `_clock llm_kernel daemon _sweeper` running；`tools ls` fsm_ann **41**、eca_bob **39**（有 grep／glob／todo／goal／subagent，eca 沒有 do_task、fork）；`hook ls` 兩隻都是 `mail_notify`、`timer` |
| 假腦 fsm_fake 問時間 | 叫 `peek_now_time`、答 `FAKE ANSWER …`；stack 底下沒有 `tools/` |
| fsm_demo 接 `fake-demo-today` | history 23 行：self_system set／get → subagent add（subagents 1/4）→ ask（kid 答 FAKE ANSWER）→ ls → rm --purge → self_system reset → todo set → **do_task start → peek_now_time → end**（那兩則換成一則 `[aos] summary: … Task "Check the time" ended: …`）→ when 接的收尾句 |
| goal（fsm_fake） | `aos-agent goal set` 後 history 多 `[aos] goal: … (reminder 1 of 10 …)`，假腦每次都叫工具＝算進展、會一直被提醒（照設計）；`goal pause` 後停 |
| 新信提醒（fsm_fake） | `aos-mailbox send` 一封，約 20 秒內多一則 `[aos] new_mail: 1 new letter from lorkhan. …`（每 1 分鐘看一次） |
| hook 例子（臨時 tmp） | `ping_me` 每 5 秒：12 秒 5 則 `tick`；`up_down` 在 `ctl stop`／`start` 各記一行（after_stop code 0、after_start） |
| **gemma（fsm_ann）** | `peek_now_time` 16 秒答 `The current time is 12:47 PM UTC …`；`knowledge_read {"op":"ls","path":"aos/"}` 5 秒答 12 項；叫它 do_task start → peek_now_time → end → 答時間，四步都照做、摺成一則 summary（41 支工具、context 4096 沒撞到） |
| **haiku（eca_bob）** | `peek_now_time` 答 `2026-11-11 12:48:11 UTC`，沒被權限擋 |
| 停樹 | `ctl rm --purge` 兩隻臨時的、`shutdown`：`4 procs asleep`，掃 `/proc` 沒有試驗場的行程 |
| 收尾 | 清兩層 stack 與總機帳本 → 背景起根、掛 fsm_ann、eca_bob（history 都 0 行），**留著跑** |
